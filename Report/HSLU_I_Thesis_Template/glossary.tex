\newglossaryentry{convolutional neural network}
{
    name={Convolutional neural network},
    description={A class of neural networks commonly used for image analysis that is relying on convolution operations to extract features from data.}
}
\newglossaryentry{googlenet}
{
    name={GoogLeNet},
    description={A deep CNN architecture introducing inception modules, which apply convolutions of multiple kernel sizes in parallel and concatenate their outputs, increasing representational capacity without a proportional increase in parameter count.}
}
\newglossaryentry{sum square error loss}
{
    name={Sum square error loss},
    description={A loss function computing the sum of squared residuals between predicted and target values, giving quadratically increasing penalty to large deviations.}
}
\newglossaryentry{isotropic Gaussian distribution}
{
    name={Isotropic Gaussian distribution},
    description={A multivariate Gaussian $\mathcal{N}(0, I)$ whose covariance matrix is a scalar multiple of the identity, implying equal variance in every direction and no correlation between dimensions.}
}
\newglossaryentry{log-likelihood}
{
    name={Log-likelihood},
    description={The logarithm of the likelihood function $\log p_\theta(x)$, used as the training objective in generative modelling. Maximising it is equivalent to minimising the KL divergence between the model distribution and the data distribution.}
}
\newglossaryentry{cosine similarity}
{
    name={Cosine similarity},
    description={A measure of alignment between two vectors defined as $\frac{u \cdot v}{\|u\|\|v\|}$, equal to the cosine of the angle between them and invariant to vector magnitude.}
}
\newglossaryentry{precision}
{
    name={Precision},
    description={The fraction of positive predictions that are correct: $\frac{TP}{TP + FP}$.}
}
\newglossaryentry{recall}
{
    name={Recall},
    description={The fraction of ground-truth positive instances that are correctly retrieved: $\frac{TP}{TP + FN}$.}
}
\newglossaryentry{clipscore}
{
    name={CLIPScore},
    description={A reference-free image--text alignment metric computed as the cosine similarity between a CLIP image embedding and a CLIP text embedding, used to assess semantic fidelity of generated images without ground-truth references.}
}
\newglossaryentry{imagenet}
{
    name={ImageNet},
    description={A large-scale image database of over 14 million images across more than 20,000 synsets, serving as the standard benchmark for image classification and representation learning.}
}
\newglossaryentry{iwildcam}
{
    name={iWildCam},
    description={A large-scale benchmark for distribution-shift robustness in wildlife camera trap classification, where train and test locations are disjoint, inducing substantial covariate shift.}
}
\newglossaryentry{macrof1}
{
    name={Macro F1},
    description={The arithmetic mean of per-class F1 scores, weighting each class equally irrespective of support. Particularly sensitive to performance on minority classes.}
}
\newglossaryentry{mobi}
{
    name={MobI},
    description={Multi-modal Object Inpainting. A diffusion-based inpainting framework extending Paint-by-Example with a 3D bounding box conditioning signal derived from both camera and LiDAR projections, enabling geometrically consistent object insertion across modalities.}
}
\newglossaryentry{fourier embeddings}
{
    name={Fourier embeddings},
    description={A positional encoding mapping low-dimensional inputs to a higher-dimensional feature space via sinusoidal basis functions, enabling neural networks to represent high-frequency spatial structure that would otherwise be suppressed by the spectral bias of MLPs.}
}
\newglossaryentry{controlnet}
{
    name={ControlNet},
    description={An adapter architecture for injecting spatial conditioning signals into a frozen diffusion model via a trainable copy of the encoder, with zero-initialised output projections ensuring the pretrained weights are unaffected at the start of training.}
}
\newglossaryentry{beta(4,1)}
{
    name={Beta(4,1)},
    description={A Beta distribution with shape parameters $\alpha=4$, $\beta=1$, concentrated near 1, producing samples strongly biased towards the upper end of the unit interval.}
}
\newglossaryentry{dvc}
{
    name={DVC},
    description={Data Version Control. A version control system for datasets and ML pipelines that stores large artefacts in remote storage while tracking only lightweight metafiles in Git, enabling reproducible experiment histories without bloating the code repository.}
}
\newglossaryentry{git}
{
    name={Git},
    description={A distributed version control system tracking file changes as a directed acyclic graph of commits, enabling non-linear development histories, branching, and reproducible snapshots of a codebase.}
}
\newglossaryentry{uv}
{
    name={uv},
    description={A Python package and project manager implemented in Rust, providing fast dependency resolution and reproducible virtual environments as a drop-in replacement for pip and venv.}
}
\newglossaryentry{wandb}
{
    name={Weights \& Biases},
    description={A machine learning experiment tracking platform that logs hyperparameters, metrics, artefacts, and system telemetry, enabling systematic comparison and reproduction of training runs.}
}
\newglossaryentry{optuna}
{
    name={Optuna},
    description={A hyperparameter optimisation framework implementing define-by-run search spaces and efficient samplers including Tree-structured Parzen Estimators and CMA-ES, with built-in pruning of unpromising trials.}
}
\newglossaryentry{m1}
{
    name={Apple M1},
    first={M1}
    description={Apple's first ARM-based system-on-chip integrating CPU, GPU, Neural Engine, and unified memory on a single die, with GPU compute exposed to frameworks via the Metal Performance Shaders backend.}
}
\newglossaryentry{labelstudio}
{
    name={Label Studio},
    description={An open-source data annotation platform supporting image, text, audio, and video labelling workflows, with export to standard formats including YOLO, COCO, and Pascal VOC.}
}
\newglossaryentry{cspdarknet53}
{
    name={CSPDarkNet53},
    description={A backbone architecture extending DarkNet-53 with Cross Stage Partial connections, which partition the feature map at each stage into two branches — one through the residual block, one bypassing it — reducing gradient redundancy while maintaining representational capacity.}
}
